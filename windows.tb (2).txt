#!/usr/bin/env python                              reducer2.py

gender, age = line.strip().split(',')

if gender not in data:

data[gender] = [0, 0]

data[gender][0] += float(age)

data[gender][1] += 1

for gender, values in data.items():

from pyspark import SparkContext

from pyspark.sql import SQLContext

# Step 1: Start Spark

sc = SparkContext()

sql = SQLContext(sc)

# Step 2: Create data

data = [

("Laptop", "Electronics", 50000),

("Phone", "Electronics", 30000),

("Rice", "Grocery", 5000),

("Oil", "Grocery", 2000),

("Chair", "Furniture", 4000),

("Table", "Furniture", 8000)

]


Step 3: Create DataFrame

columns = ["Product", "Category", "Sales"]

df = sql.createDataFrame(data, columns)

# Step 4: Display all products

print("All Products:")

df.show()

# Step 5: Filter sales above 5000

print("Products with sales above 5000:")

df.filter(df.Sales > 5000).show()


Step 6: Total sales by category

print("Total Sales by Category:")

df.groupBy("Category").sum("Sales").show()

# Step 7: Find the highest-selling product

print("Highest-Selling Product:")

df.orderBy(df.Sales.desc()).show(1)

# Step 8: Sort sales from highest to lowest

print("Products sorted by Sales:")

df.orderBy(df.Sales.desc()).show()

# Step 9: Stop Spark

sc.stop()

from pyspark import SparkContext

from pyspark.sql import SQLContext

# Step 1: Start Spark

sc = SparkContext()

sql = SQLContext(sc)

# Step 2: Create data

data = [

("Anu", "CSE", 85),

("Bala", "AIML", 90),

("Charan", "IT", 78),

("Deepa", "ECE", 88),

("Esha", "CSE", 92),

("Farhan", "AIML", 81)

]

# Step 3: Create DataFrame

columns = ["Name", "Department", "Marks"]

df = sql.createDataFrame(data, columns)

# Step 4: Show the full DataFrame

print("Full Student Data:")

df.show()

# Step 5: Filter students who scored above 85

print("Students who scored above 85:")

df.filter(df.Marks > 85).show()

# Step 6: Group by department and find average marks

print("Average Marks per Department:")

df.groupBy("Department").avg("Marks").show()

# Step 7: Sort by marks (descending)

print("Students sorted by Marks (highest first):")

df.orderBy(df.Marks.desc()).show()

# Step 8: Stop Spark

sc.stop()
